find_package(Qt6 6.5 REQUIRED COMPONENTS Core Concurrent Network)
find_package(ZLIB REQUIRED)
include(../cmake/recording.cmake)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${recording_config}")
add_library(stars_net STATIC
    adsb.cpp adsbcache.cpp adsbhttp.cpp adsbindex.cpp adsbtrace.cpp
    trajectory.cpp trajectory.h track.cpp track.h archive.cpp archive.h tais.cpp tais.h
)
target_compile_definitions(stars_net PRIVATE STARS_ADSB_RETENTION_DAYS=${retention_days})
target_compile_features(stars_net PUBLIC cxx_std_20)
target_include_directories(stars_net PUBLIC "${PROJECT_SOURCE_DIR}")
target_link_libraries(stars_net PUBLIC Qt6::Core Qt6::Concurrent Qt6::Network PRIVATE stars_zstd ZLIB::ZLIB)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(stars_adsb_tests tests/adsb.cpp)
    set_target_properties(stars_adsb_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(stars_adsb_tests PRIVATE stars_net Qt6::Test stars_zstd ZLIB::ZLIB)
    add_test(NAME stars_adsb COMMAND stars_adsb_tests)
    qt_add_executable(stars_trajectory_tests tests/trajectory.cpp)
    set_target_properties(stars_trajectory_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(stars_trajectory_tests PRIVATE stars_net Qt6::Test)
    add_test(NAME stars_trajectory COMMAND stars_trajectory_tests)
    qt_add_executable(stars_tais_tests tests/tais.cpp)
    set_target_properties(stars_tais_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(stars_tais_tests PRIVATE stars_net Qt6::Test stars_zstd)
    add_test(NAME stars_tais COMMAND stars_tais_tests)
endif()
add_executable(taischeck EXCLUDE_FROM_ALL check.cpp)
target_link_libraries(taischeck PRIVATE stars_net)

add_executable(adsbcheck EXCLUDE_FROM_ALL adsbcheck.cpp)
target_link_libraries(adsbcheck PRIVATE stars_net)
