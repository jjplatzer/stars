find_package(Qt6 6.5 REQUIRED COMPONENTS Core Concurrent)
add_library(stars_net STATIC track.cpp track.h archive.cpp archive.h tais.cpp tais.h)
target_compile_features(stars_net PUBLIC cxx_std_20)
target_include_directories(stars_net PUBLIC "${PROJECT_SOURCE_DIR}")
target_link_libraries(stars_net PUBLIC Qt6::Core Qt6::Concurrent PRIVATE stars_zstd)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(stars_tais_tests tests/tais.cpp)
    set_target_properties(stars_tais_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(stars_tais_tests PRIVATE stars_net Qt6::Test stars_zstd)
    add_test(NAME stars_tais COMMAND stars_tais_tests)
endif()

target_link_libraries(taischeck PRIVATE stars_net)
